cmake_minimum_required(VERSION 3.16)
project(HynisBridgeTests C)
enable_testing()
add_executable(gl4metal_session_test gl4metal_session_test.c)
target_include_directories(gl4metal_session_test PRIVATE ../Natives/ctxbridges)
set_target_properties(gl4metal_session_test PROPERTIES C_STANDARD 99)
if(MSVC)
    target_compile_options(gl4metal_session_test PRIVATE /UNDEBUG)
else()
    target_compile_options(gl4metal_session_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
add_test(NAME gl4metal_session COMMAND gl4metal_session_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME gl4metal_library COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_gl4metal_library.py)
